cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD waveshare_rp2350_usb_a CACHE STRING "Target board")
if(NOT PICO_BOARD STREQUAL "waveshare_rp2350_usb_a")
  message(FATAL_ERROR "The reader requires Waveshare RP2350-USB-A")
endif()
if(NOT PICO_SDK_PATH)
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT PICO_PIO_USB_PATH)
  set(PICO_PIO_USB_PATH "$ENV{PICO_PIO_USB_PATH}" CACHE PATH "Pico-PIO-USB checkout")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(ch552_reader VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ch552_reader main.c reader.c probe_protocol.c board_status.c
  result_disk.c result_drive.c usb_descriptors.c)
target_include_directories(ch552_reader PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(ch552_reader PRIVATE READER_VERSION="${PROJECT_VERSION}"
  PIO_USB_DP_PIN_DEFAULT=12 PICO_STACK_SIZE=0x1000)
target_compile_options(ch552_reader PRIVATE -Wall -Wextra -Werror -fstack-usage)
target_link_libraries(ch552_reader PRIVATE pico_stdlib pico_unique_id hardware_pio
  tinyusb_device tinyusb_host tinyusb_pico_pio_usb)
pico_generate_pio_header(ch552_reader "${PICO_SDK_PATH}/src/rp2_common/pico_status_led/ws2812.pio")
pico_enable_stdio_uart(ch552_reader 0)
pico_enable_stdio_usb(ch552_reader 0)
pico_set_program_name(ch552_reader "CH552 Read-Only Bootloader Reader")
pico_set_program_version(ch552_reader "${PROJECT_VERSION}")
pico_add_extra_outputs(ch552_reader)
